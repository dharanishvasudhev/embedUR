#!/bin/bash

LOG_FILE="errors.log"

search_recursive() {
    local dir="$1"
    local keyword="$2"

    for item in "$dir"/*; do
        if [[ -d "$item" ]]; then
            search_recursive "$item" "$keyword"
        elif [[ -f "$item" ]]; then
            if grep -q "$keyword" "$item" 2>>"$LOG_FILE"; then
                echo "Match found in: $item"
            fi
        fi
    done
}

while [[ $# -gt 0 ]]; do
    case "$1" in
        -d) DIR="$2"; shift 2 ;;
        -k) KEYWORD="$2"; shift 2 ;;
        -f) FILE="$2"; shift 2 ;;
        --help) show_help; exit 0 ;;
        *) echo "Invalid argument: $1" | tee -a "$LOG_FILE"; exit 1 ;;
    esac
done

[[ ! "$KEYWORD" =~ ^[a-zA-Z0-9_]+$ ]] && \
echo "Invalid keyword" | tee -a "$LOG_FILE" && exit 1

if [[ -n "$FILE" ]]; then
    if [[ -f "$FILE" ]]; then
        CONTENT=$(cat "$FILE")
        if grep -q "$KEYWORD" <<< "$CONTENT"; then
            echo "Keyword found in file: $FILE"
        fi
        echo "Exit status: $?"
    else
        echo "File not found: $FILE" | tee -a "$LOG_FILE"
    fi
fi

if [[ -n "$DIR" ]]; then
    if [[ -d "$DIR" ]]; then
        search_recursive "$DIR" "$KEYWORD"
        echo "Exit status: $?"
    else
        echo "Directory not found: $DIR" | tee -a "$LOG_FILE"
    fi
fi
